class Solution {
    public String mapWordWeights(String[] words, int[] weights) {
        StringBuilder sb = new StringBuilder();
        for(String s : words) {
            int weight = 0;
            for(char ch: s.toCharArray()){
                weight += weights[(int)ch - 97]; 
            }
            sb.append((char)(122 - weight % 26));
        }
        return sb.toString();
    }
}
