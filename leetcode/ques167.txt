class Solution {
    public int[] twoSum(int[] nums, int target) {
        int left, right, mid;
        for(int i = 0; i < nums.length; i++){
            int req = target - nums[i];
            left = 0; right = nums.length;
            while(left < right){
                mid = (left+right) / 2;
                if(req == nums[mid] && mid != i ) return new int[]{i + 1, mid + 1};
                else if(req < nums[mid]) {
                    right = mid;
                } else left = mid + 1;
            }
        }
        return new int[]{-1, -1};
    }
}
