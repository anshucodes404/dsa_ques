class Solution {
    public void moveZeroes(int[] nums) {
        int left = 0; // a pointer that points to index till which no non-zero elements are there

        for(int i = 0; i < nums.length; i++){
            if(nums[i] == 0){
                continue;
            }
            swap(left, i, nums);
            left++;
        }
    }

    private void swap(int left, int right, int[] arr){
        int temp = arr[left];
        arr[left] = arr[right];
        arr[right] = temp;
    }
}
