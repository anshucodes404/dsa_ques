class Solution {
    public int findMin(int[] nums) {
        int pivot = pivotFind(nums);
        if(pivot == -1) return nums[0];
        else return nums[pivot + 1];
    }

    private int pivotFind(int[] arr) {
        int s = 0;
        int e = arr.length - 1;
        int m;
        while(s <= e){
            m = (s + e) / 2;
            if(m < e && arr[m] > arr[m + 1]) return m;
            if(m > s && arr[m] < arr[m - 1]) return m - 1;
            if(arr[m] >= arr[s]) s = m + 1;
            else e = m - 1;
        }
        return -1;
    }
}
