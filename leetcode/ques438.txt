class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> list = new ArrayList<>();
        int[] arr1 = new int[26];
        int[] arr2 = new int[26];

        for (char ch : p.toCharArray()) {
            arr1[ch - 'a']++;
        }

        int window = p.length();
        for (int i = 0; i < s.length(); i++) {
            arr2[s.charAt(i) - 'a']++;

            if (i >= window) {
                arr2[s.charAt(i - window) - 'a']--;
            }

            if (isSame(arr1, arr2))
                list.add(i - window + 1);
        }
        return list;

    }

    private boolean isSame(int[] arr1, int[] arr2) {
        for (int i = 0; i < 26; i++) {
            if (arr1[i] != arr2[i])
                return false;
        }
        return true;
    }
}
