class Solution {
    public int[] xorQueries(int[] arr, int[][] q) {
        int[] xArr = new int[arr.length];
        int[] ans = new int[q.length];

        int x = 0;

        for(int i = 0; i < arr.length; i++){
            x ^= arr[i];
            xArr[i] = x;
        }

        for(int i = 0; i < q.length; i++){
            int res = xArr[q[i][1]] ^ xArr[q[i][0]] ^ arr[q[i][0]];
            ans[i] = res;
        }
        return ans;
    }
}
