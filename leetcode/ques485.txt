class Solution {
    public int findMaxConsecutiveOnes(int[] nums) {
        int maxlen = 0;
        int len = 0;
        for (int num : nums) {
            if (num == 1) {
                len++;
                maxlen = Math.max(len, maxlen);
            } else {
                len = 0;
            }
        }
        return maxlen;
    }
}
