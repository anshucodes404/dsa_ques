class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int maxsum = 0;
        for(int i = 0; i < k; i++) maxsum += nums[i]; // initial window sum

        int sum = maxsum;
        int i =0;
        while(i + k < nums.length){
            sum = sum - nums[i] + nums[i+k];
            maxsum = Math.max(sum, maxsum);
            i++;
        }

        return (double)maxsum / k;

    }
}
